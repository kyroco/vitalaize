# Remove VitalAIze, keeping the database and settings.

wait 40 "The board is running"
press "Remove VitalAIze…"
wait 5 "Remove VitalAIze from this Mac?"
snap 13-remove-sheet
click "Cancel"
gone 5 "Remove VitalAIze from this Mac?"
expect "The board is running"
press "Remove VitalAIze…"
wait 5 "Remove VitalAIze from this Mac?"
press "Remove"
wait 60 "VitalAIze was removed from this Mac"
expect "Stopping the board"
expect "Its database and settings are still in"
snap 14-removed
click "Set up again"
wait 90 "What this Mac does"
