# Remove VitalAIze and delete its database and settings too.

wait 40 "The board is running"
press "Remove VitalAIze…"
wait 5 "Remove VitalAIze from this Mac?"
toggle "Also delete the database and settings"
snap 15-remove-sheet-delete
press "Remove"
wait 60 "VitalAIze was removed from this Mac"
expect "Its database and settings were deleted too, and any key typed in Settings was taken out of the keychain."
snap 15-removed-all
click "Set up again"
wait 90 "What this Mac does"
