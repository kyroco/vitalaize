# The first screen of a Mac that runs the board, and its Settings.

wait 40 "The board is running"
expect "Hub and collector"
absent "Mended"
absent "Restarting"
snap 08-status
click "Check again"
wait 10 "The board is running"
click "Open the board"
opened "http://localhost:{PORT}/"
click "Show the log"
opened "board.log"
click "Restart the board"
wait 5 "Restarting the board…"
disabled "Restart the board"
snap 09-restarting
wait 90 "The board restarted and is answering."
wait 20 "The board is running"
enabled "Restart the board"
snap 09-restarted
